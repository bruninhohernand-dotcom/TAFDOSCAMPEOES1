async function analyzeLatestEdital() { const btn = document.getElementById('btnAnalyze'); try { if (!session) { const { data, error } = await sb.auth.getSession(); if (error) throw error; session = data.session; } if (!session?.access_token) { toast('Sua sessão expirou. Entre novamente na sua conta.'); go('auth'); return; } const aluno = await ensureStudent(); if (!aluno) { toast('Não foi possível localizar seu cadastro.'); return; } btn.disabled = true; btn.textContent = 'Analisando edital...'; const { data: editais, error: editalError } = await sb .from('editais') .select('*') .eq('aluno_id', aluno.id) .order('criado_em', { ascending: false }) .limit(1); if (editalError) throw editalError; const edital = editais?.[0]; if (!edital) { toast('Nenhum edital foi encontrado. Envie o PDF na avaliação.'); return; } const { data, error } = await sb.functions.invoke( 'analisar-edital', { body: { edital_id: edital.id }, headers: { Authorization: `Bearer ${session.access_token}` } } ); if (error) { console.error('Erro analisar-edital:', error); throw error; } console.log('Análise do edital:', data); toast('Edital analisado com sucesso!'); await loadStudentArea(); } catch (error) { console.error('analisar-edital:', error); toast( 'Não foi possível analisar o edital: ' + (error?.message || 'erro desconhecido') ); } finally { if (btn) { btn.disabled = false; btn.textContent = 'Analisar meu edital'; } } }
